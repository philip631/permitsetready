// ---------------------------------------------------------------- entry points
function doGet(e) {
  return json_({ok:true, service:'PermitSetReady', version:VERSION});
}

function doPost(e) {
  try {
    if (e && e.parameter && e.parameter.hook === 'ls') return json_(lsWebhook_(e));
    var req = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    var out = route_(req);
    return json_(Object.assign({ok:true}, out || {}));
  } catch (err) {
    var userErr = /log in|Not allowed|code is not right|valid email|Too many|not waiting|balance is due|Unknown action/i.test(String(err && err.message));
    log_(userErr ? 'info' : 'error', 'doPost', userErr ? String(err.message) : String(err && err.stack || err));
    return json_({ok:false, error:String(err && err.message || err)});
  }
}

function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

function route_(req) {
  var a = req.action;
  if (a === 'auth.start') return authStart_(req.email);
  if (a === 'auth.verify') return authVerify_(req.email, req.code);
  if (a && a.indexOf('runner.') === 0) return runner_(req);
  var s = session_(req.token);
  if (a === 'auth.me') return {user:userOf_(s)};
  if (a === 'auth.logout') { endSession_(req.token); return {}; }
  if (a === 'db.get') return dbGet_(s, req.path);
  if (a === 'db.set') return dbSet_(s, req.path, req.data, !!req.merge);
  if (a === 'db.delete') return dbDelete_(s, req.path);
  if (a === 'db.list') return dbList_(s, req.path, req.orderBy, req.dir, req.limit);
  if (a === 'pay.checkout') return payCheckout_(s, req.path, req.kind);
  if (a === 'files.deliverables') return deliverables_(s, req.path);
  if (a === 'files.download') return download_(s, req.path, req.fileId);
  if (a === 'admin.lists') { needAdmin_(s); return {reviewers:rows_('Reviewers'), consultants:rows_('Consultants')}; }
  throw new Error('Unknown action');
}

// ---------------------------------------------------------------- auth (email + 6-digit code)
function authStart_(email) {
  email = String(email || '').trim().toLowerCase();
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) throw new Error('Enter a valid email address.');
  var cache = CacheService.getScriptCache(), k = 'code:' + hash_(email);
  var tries = Number(cache.get('tries:' + k) || 0);
  if (tries >= 5) throw new Error('Too many codes requested. Try again in 15 minutes.');
  cache.put('tries:' + k, String(tries + 1), 900);
  var code = String(Math.floor(100000 + Math.random() * 900000));
  cache.put(k, hash_(code), 600);
  var html = shell_('Your login code',
    '<p style="margin:0 0 14px">Use this code to open your PermitSetReady project page. It expires in 10 minutes.</p>' +
    '<p style="font-size:34px;font-weight:bold;letter-spacing:8px;font-family:\'Courier New\',monospace;margin:8px 0 18px;color:#0E1F36">' + code + '</p>' +
    '<p style="margin:0;color:#5B6B7F;font-size:13px">If you didn\'t ask for this, you can ignore this email.</p>');
  // Login codes always send: a client can't log in otherwise.
  GmailApp.sendEmail(email, 'Your PermitSetReady login code: ' + code, 'Your PermitSetReady login code is ' + code + '. It expires in 10 minutes.', fromOpts_({htmlBody:html}));
  return {sent:true};
}

function authVerify_(email, code) {
  email = String(email || '').trim().toLowerCase();
  var cache = CacheService.getScriptCache(), k = 'code:' + hash_(email);
  var want = cache.get(k);
  if (!want || want !== hash_(String(code || '').trim())) throw new Error('That code is not right or has expired. Request a new one.');
  cache.remove(k);
  var token = Utilities.getUuid() + Utilities.getUuid();
  var uid = 'u_' + hash_(email).slice(0, 20);
  var exp = new Date(Date.now() + 30 * 24 * 3600 * 1000);
  sh_('Sessions').appendRow([hash_(token), email, uid, new Date(), exp]);
  return {token:token, user:userOf_({email:email, uid:uid})};
}

function session_(token) {
  if (!token) throw new Error('Please log in.');
  var cache = CacheService.getScriptCache(), th = hash_(token), c = cache.get('s:' + th);
  if (c) return JSON.parse(c);
  var v = sh_('Sessions').getDataRange().getValues();
  for (var i = v.length - 1; i >= 1; i--) {
    if (v[i][0] === th) {
      if (new Date(v[i][4]) < new Date()) break;
      var s = {email:v[i][1], uid:v[i][2]};
      s.admin = isAdmin_(s.email); s.reviewer = isReviewer_(s.email);
      cache.put('s:' + th, JSON.stringify(s), 600);
      return s;
    }
  }
  throw new Error('Your login has expired. Please log in again.');
}
function endSession_(token) {
  if (!token) return; var th = hash_(token); CacheService.getScriptCache().remove('s:' + th);
  var sh = sh_('Sessions'), v = sh.getDataRange().getValues();
  for (var i = v.length - 1; i >= 1; i--) if (v[i][0] === th) sh.getRange(i + 1, 5).setValue(new Date(0));
}

function isReviewer_(email) { return rows_('Reviewers').some(function(r){ return String(r.Email).toLowerCase() === String(email).toLowerCase() && r.Active !== false && String(r.Active).toUpperCase() !== 'FALSE'; }); }
function userOf_(s) { var adm = isAdmin_(s.email); return {id:s.uid, email:s.email, name:'', isAdmin:adm || isReviewer_(s.email), role:adm ? 'admin' : (isReviewer_(s.email) ? 'reviewer' : 'client')}; }

// ---------------------------------------------------------------- document store (Firestore-like paths)
function parts_(path) { path = String(path || '').replace(/^\/+|\/+$/g, ''); if (!path || /\.\./.test(path)) throw new Error('Bad path'); return path.split('/'); }
function canRead_(s, p) {
  if (s.admin || s.reviewer) return true;
  if (p[0] === 'settings') return true;
  return p[0] === 'intakes' && p[1] === s.uid;
}
function canWrite_(s, p) {
  if (s.admin || s.reviewer) return true;
  return p[0] === 'intakes' && p[1] === s.uid;
}
function findRow_(path) {
  var sh = sh_('Docs'); if (sh.getLastRow() < 2) return 0;
  var f = sh.getRange(2, 1, sh.getLastRow() - 1, 1).createTextFinder(path).matchEntireCell(true).findNext();
  return f ? f.getRow() : 0;
}
function readRow_(row) {
  var v = sh_('Docs').getRange(row, 1, 1, 4 + DOC_CHUNKS).getValues()[0];
  return JSON.parse(v.slice(4).join('') || '{}');
}
function rawGet_(path) { var r = findRow_(path); return r ? readRow_(r) : null; }
function rawSet_(path, data) {
  var str = JSON.stringify(data), chunks = [];
  for (var i = 0; i < DOC_CHUNKS; i++) chunks.push(str.slice(i * DOC_CHUNK, (i + 1) * DOC_CHUNK));
  if (str.length > DOC_CHUNK * DOC_CHUNKS) throw new Error('Record too large');
  var p = path.split('/'), row = [path, p.slice(0, -1).join('/'), p[p.length - 1], data.updatedAt || now_()].concat(chunks);
  var sh = sh_('Docs'), r = findRow_(path);
  if (r) sh.getRange(r, 1, 1, row.length).setValues([row]); else sh.appendRow(row);
}
function rawDelete_(path) { var r = findRow_(path); if (r) sh_('Docs').deleteRow(r); }

function dbGet_(s, path) {
  var p = parts_(path); if (!canRead_(s, p)) throw new Error('Not allowed.');
  var d = rawGet_(p.join('/'));
  if (d && isFilePath_(p)) d = withDataUrl_(d);
  return {exists:!!d, data:d};
}

function dbList_(s, path, orderBy, dir, limit) {
  var p = parts_(path); if (!canRead_(s, p)) throw new Error('Not allowed.');
  var parent = p.join('/'), sh = sh_('Docs'), out = [];
  if (sh.getLastRow() >= 2) {
    var v = sh.getRange(2, 1, sh.getLastRow() - 1, 4 + DOC_CHUNKS).getValues();
    v.forEach(function(r){ if (r[1] === parent) { var d = JSON.parse(r.slice(4).join('') || '{}'); out.push({id:r[2], data:d}); } });
  }
  if (orderBy) out.sort(function(a, b){ var x = a.data[orderBy] || '', y = b.data[orderBy] || ''; return (x < y ? -1 : x > y ? 1 : 0) * (dir === 'desc' ? -1 : 1); });
  if (limit) out = out.slice(0, limit);
  if (isFilePath_(p.concat(['x']))) out = out.map(function(o){ return {id:o.id, data:withDataUrl_(o.data)}; });
  return {docs:out};
}

function dbSet_(s, path, data, merge) {
  var p = parts_(path); if (!canWrite_(s, p)) throw new Error('Not allowed.');
  path = p.join('/');
  var lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    var prev = rawGet_(path);
    if (merge && prev) data = deepMerge_(JSON.parse(JSON.stringify(prev)), data);
    if (isFilePath_(p)) { data = saveFile_(p, data, prev); rawSet_(path, data); if (data.kind === 'comments') onComments_(p); return {}; }
    if (isProjectPath_(p)) data = guardProject_(s, data, prev);
    if (p[0] === 'settings' && !s.admin) throw new Error('Not allowed.');
    rawSet_(path, data);
    if (isProjectPath_(p)) onProjectSaved_(p, data, prev);
    return {};
  } finally { lock.releaseLock(); }
}

function dbDelete_(s, path) {
  var p = parts_(path); if (!canWrite_(s, p)) throw new Error('Not allowed.');
  var d = rawGet_(p.join('/'));
  if (d && d.driveId) { try { DriveApp.getFileById(d.driveId).setTrashed(true); } catch (e) {} }
  rawDelete_(p.join('/'));
  return {};
}

function deepMerge_(a, b) {
  Object.keys(b || {}).forEach(function(k){
    if (b[k] && typeof b[k] === 'object' && !Array.isArray(b[k]) && a[k] && typeof a[k] === 'object' && !Array.isArray(a[k])) a[k] = deepMerge_(a[k], b[k]); else a[k] = b[k];
  });
  return a;
}
function isProjectPath_(p) { return p.length === 4 && p[0] === 'intakes' && p[2] === 'projects'; }
function isFilePath_(p) { return p.length === 6 && p[0] === 'intakes' && p[2] === 'projects' && p[4] === 'files'; }
function projPath_(uid, pid) { return 'intakes/' + uid + '/projects/' + pid; }

/** Clients can edit their answers but never server-owned fields. */
function guardProject_(s, data, prev) {
  prev = prev || {};
  if (s.admin || s.reviewer) return data;
  if (CLIENT_STATUSES.indexOf(data.status) < 0) data.status = prev.status || 'draft';
  if (prev.status && CLIENT_STATUSES.indexOf(prev.status) < 0) data.status = prev.status; // locked once paid
  data.admin = prev.admin || {};
  data.server = prev.server || {};
  var pay = prev.payment || {};
  data.payment = Object.assign({}, data.payment || {}, {deposit:pay.deposit || null, balance:pay.balance || null});
  return data;
}

function saveFile_(p, data, prev) {
  if (!data.dataUrl) return data;
  var m = /^data:([^;]+);base64,(.*)$/.exec(data.dataUrl); if (!m) throw new Error('Bad file');
  var proj = rawGet_(p.slice(0, 4).join('/')) || {};
  var folder = proj.server && proj.server.folderId ? subfolder_(DriveApp.getFolderById(proj.server.folderId), kindFolderName_(data.kind)) : subfolder_(stagingFolder_(), p[1] + '_' + p[3]);
  var date = Utilities.formatDate(new Date(data.createdAt || Date.now()), TZ, 'yyMMdd');
  var name = date + ' - ' + kindLabel_(data.kind) + ' - ' + String(data.name || 'file').replace(/[\\\/]/g, '_');
  var file = folder.createFile(Utilities.newBlob(Utilities.base64Decode(m[2]), m[1], name));
  if (prev && prev.driveId) { try { DriveApp.getFileById(prev.driveId).setTrashed(true); } catch (e) {} }
  var out = Object.assign({}, data); delete out.dataUrl;
  out.driveId = file.getId(); out.size = file.getSize(); out.staged = !(proj.server && proj.server.folderId);
  return out;
}
function withDataUrl_(d) {
  if (!d || !d.driveId) return d;
  try {
    var b = DriveApp.getFileById(d.driveId).getBlob();
    return Object.assign({}, d, {dataUrl:'data:' + b.getContentType() + ';base64,' + Utilities.base64Encode(b.getBytes())});
  } catch (e) { return d; }
}
function moveStagedFiles_(uid, pid, folder) {
  var list = dbListRaw_(projPath_(uid, pid) + '/files');
  list.forEach(function(o){
    if (!o.data.driveId || !o.data.staged) return;
    try { DriveApp.getFileById(o.data.driveId).moveTo(subfolder_(folder, kindFolderName_(o.data.kind))); o.data.staged = false; rawSet_(projPath_(uid, pid) + '/files/' + o.id, o.data); } catch (e) { log_('warn', 'moveStaged', e); }
  });
}
function dbListRaw_(parent) {
  var sh = sh_('Docs'), out = []; if (sh.getLastRow() < 2) return out;
  sh.getRange(2, 1, sh.getLastRow() - 1, 4 + DOC_CHUNKS).getValues().forEach(function(r){ if (r[1] === parent) out.push({id:r[2], data:JSON.parse(r.slice(4).join('') || '{}')}); });
  return out;
}

function deliverables_(s, path) {
  var p = parts_(path); if (!canRead_(s, p) || !isProjectPath_(p)) throw new Error('Not allowed.');
  var d = rawGet_(p.join('/')) || {};
  if (!(s.admin || s.reviewer) && !(d.server && d.server.released)) return {files:[], released:false};
  if (!d.server || !d.server.folderId) return {files:[], released:!!(d.server && d.server.released)};
  var pdf = subfolder_(DriveApp.getFolderById(d.server.folderId), 'PDF'), it = pdf.getFiles(), out = [];
  while (it.hasNext()) { var f = it.next(); if (!f.isTrashed()) out.push({id:f.getId(), name:f.getName(), size:f.getSize()}); }
  return {files:out, released:true};
}
function download_(s, path, fileId) {
  var list = deliverables_(s, path).files;
  if (!list.some(function(f){ return f.id === fileId; })) throw new Error('Not allowed.');
  var b = DriveApp.getFileById(fileId).getBlob();
  return {name:b.getName(), dataUrl:'data:' + b.getContentType() + ';base64,' + Utilities.base64Encode(b.getBytes())};
}

// ---------------------------------------------------------------- pricing (same rules as the questionnaire)
function scopesOf_(it) { var a = Array.isArray(it.scopes) ? it.scopes : (it.scope ? [it.scope] : []); return SCOPE_ORDER.filter(function(k){ return a.indexOf(k) >= 0; }); }
function has_(it, k) { return scopesOf_(it).indexOf(k) >= 0; }
function priceOf_(it) {
  var t = it.tier || 't2', sc = scopesOf_(it), p = PRICES[t] || PRICES.t2;
  if (t === 't1') return PRICES.t1[sc[0]] || 59;
  return sc.reduce(function(sum, k, i){
    var base = p[k] || 0;
    if (k === 'unit' && t === 't3' && !has_(it, 'bge') && it.basement && it.basement.entrance === 'Only through the main house') base = PRICES.t3.unitNew;
    return sum + (i === 0 ? base : Math.round(base * PRICES.addonRate / 50) * 50);
  }, 0);
}
function needsSurvey_(it) { return has_(it,'garage') || has_(it,'shed') || has_(it,'bge') || has_(it,'deck'); }
function measEst_(it) {
  var m = it.measure || {}, sq = Number(m.sqft) || 0; if (!(sq > 0)) return 0;
  var f = m.feat || {}, site = needsSurvey_(it) || !!(f.shed || f.deck || f.patio || f.pool || f.garage || f.other);
  var km = MEAS_KM[(it.property || {}).municipality] || 60;
  var sub = MEAS.base + Math.max(0, sq - MEAS.freeSqft) * MEAS.perSqft + ((has_(it,'bge') ? 1 : 0) + (has_(it,'deck') ? 1 : 0)) * MEAS.elev + MEAS.roof + km * 2 * MEAS.perKm + (site ? MEAS.site : 0);
  return Math.ceil(sub * (1 + MEAS.markup) / 100) * 100;
}
function packageName_(it) { return (TIERS[it.tier] || 'Project') + ' · ' + (scopesOf_(it).map(function(k){ return SCOPES[k]; }).join(' + ') || 'Project'); }
function money_(n) { return '$' + Number(n || 0).toLocaleString('en-CA'); }

// ---------------------------------------------------------------- project lifecycle
function onProjectSaved_(p, d, prev) {
  var uid = p[1], pid = p[3], path = p.join('/');
  var before = (prev && prev.status) || 'draft', after = d.status || 'draft';
  var changed = false;
  d.server = d.server || {};
  // submitted: give it a PSR number, a Drive folder, an intake summary, and a "received" email
  if (after !== 'draft' && after !== 'declined' && d.tier !== 't1' && !d.server.folderId) {
    var made = createJobFolder_(d);
    d.server.psr = made.psr; d.server.folderId = made.id; d.server.folderUrl = made.url; d.server.submittedAt = now_();
    moveStagedFiles_(uid, pid, DriveApp.getFolderById(made.id));
    writeSummary_(d, made.id, path);
    changed = true;
    queueEmail_('received', d, path);
  }
  if (before !== after) {
    d.server.history = (d.server.history || []).concat([{status:after, at:now_()}]);
    changed = true;
    onStatus_(after, d, path);
  }
  if (changed) rawSet_(path, d);
  upsertJob_(d, path);
  syncIndex_(uid, d);
}

function setStatus_(path, status, extra) {
  var d = rawGet_(path); if (!d) throw new Error('No such project');
  var prev = JSON.parse(JSON.stringify(d));
  d.status = status; d.updatedAt = now_();
  if (extra) d = deepMerge_(d, extra);
  rawSet_(path, d);
  onProjectSaved_(path.split('/'), d, prev);
  return d;
}

function onStatus_(status, d, path) {
  if (status === 'confirmed') {
    if (!d.server.depositEmailed) { queueEmail_('deposit', d, path); d.server.depositEmailed = true; }
    if (d.measure && d.measure.want && !(d.server && d.server.asbuiltReceived)) {
      queueEmail_('measuring_partner', d, path); queueEmail_('measuring_client', d, path);
      // move straight on to the measuring stage
      d.status = 'measuring'; d.server.history = (d.server.history || []).concat([{status:'measuring', at:now_()}]);
    }
  }
  if (status === 'review') {
    var r = pickReviewer_(d); d.admin = d.admin || {}; d.admin.reviewer = r.email;
    sendInternal_(r.email, 'PermitSetReady: ' + (d.server.psr || '') + ' ready for review',
      '<p>' + esc_(d.server.psr || '') + ' · ' + esc_(packageName_(d)) + ' · ' + esc_((d.property || {}).address || '') + '</p>' +
      '<p>The drawings and the check report are in the job folder: <a href="' + d.server.folderUrl + '">open folder</a>.</p>' +
      '<p><a href="' + site_() + '/start/#admin">Open the admin page</a> to approve or send it back with notes.</p>');
  }
  if (status === 'consultant') {
    var c = pickConsultant_(d.admin && d.admin.consultant, 'structural');
    if (c) { d.admin.consultant = c.Name; queueEmail_('consultant_transmittal', d, path, c); }
    queueEmail_('consultant_client', d, path);
  }
  if (status === 'ready') {
    if (d.payment && d.payment.balance) { // a revision for City comments: already paid, release straight away
      d.status = 'delivered'; d.server.released = true; d.server.history = (d.server.history || []).concat([{status:'delivered', at:now_()}]);
      queueEmail_('revised', d, path);
    } else queueEmail_('ready', d, path);
  }
  if (status === 'delivered') { d.server.released = true; queueEmail_('delivered', d, path); }
  if (status === 'revising') sendInternal_(digestTo_(), 'PermitSetReady: City comments uploaded for ' + (d.server.psr || ''), '<p>The client uploaded the examiner\'s comments. The drafting runner will pick up the revision.</p><p><a href="' + (d.server.folderUrl || '#') + '">Job folder</a></p>');
}

}

function createJobFolder_(d) {
  var jobs = DriveApp.getFolderById(prop_('JOBS_FOLDER_ID'));
  var yy = Utilities.formatDate(new Date(), TZ, 'yy'), max = 0, it = jobs.getFolders();
  while (it.hasNext()) { var m = /^PSR-(\d\d)(\d\d)\b/.exec(it.next().getName()); if (m && m[1] === yy) max = Math.max(max, Number(m[2])); }
  rows_('Jobs').forEach(function(r){ var m = /^PSR-(\d\d)(\d\d)$/.exec(r.PSR); if (m && m[1] === yy) max = Math.max(max, Number(m[2])); });
  var psr = 'PSR-' + yy + ('0' + (max + 1)).slice(-2);
  var c = d.contact || {}, pr = d.property || {};
  var name = psr + ' - ' + (c.name || 'Client') + ' - ' + (pr.address || 'Address') + (pr.municipality ? ', ' + pr.municipality : '');
  var folder = jobs.createFolder(name.replace(/[\\\/]/g, '-'));
  var tpl = prop_('TEMPLATE_FOLDER_ID'); if (tpl) { try { copyTree_(DriveApp.getFolderById(tpl), folder); } catch (e) { log_('warn', 'copyTree', e); } }
  ['Received','Photos','Survey','PDF','Consultant'].forEach(function(n){ subfolder_(folder, n); });
  return {psr:psr, id:folder.getId(), url:folder.getUrl()};
}
function copyTree_(src, dest) {
  var fs = src.getFiles(); while (fs.hasNext()) { var f = fs.next(); if (/\.\d{4}\.rvt$/i.test(f.getName())) continue; f.makeCopy(f.getName(), dest); }
  var ds = src.getFolders(); while (ds.hasNext()) { var s = ds.next(); copyTree_(s, subfolder_(dest, s.getName())); }
}

function flatten_(o, pre, out, skip) {
  Object.keys(o || {}).forEach(function(k){
    if (!pre && skip.indexOf(k) >= 0) return;
    var v = o[k], key = pre ? pre + '.' + k : k;
    if (v && typeof v === 'object' && !Array.isArray(v)) flatten_(v, key, out, skip);
    else if (v !== '' && v !== null && v !== undefined && v !== false) out.push('  ' + key + ': ' + (Array.isArray(v) ? v.join(', ') : v));
  });
}

function upsertJob_(d, path) {
  var sh = sh_('Jobs'), c = d.contact || {}, pr = d.property || {}, s = d.server || {}, a = d.admin || {};
  if (!s.psr && d.tier !== 't1' && (d.status || 'draft') === 'draft') return; // drafts stay out of the job list
  var price = priceOf_(d);
  var row = [s.psr || '', c.name || '', c.email || '', c.phone || '', pr.address || '', pr.municipality || '', packageName_(d), price, Math.round(price * DEPOSIT_RATE),
    (d.measure && d.measure.want) ? measEst_(d) : '', STATUS_LABEL[d.status] || d.status, a.reviewer || '', a.consultant || '', a.needsEngineer ? 'Yes' : '', s.folderUrl || '', new Date(), path];
  var last = sh.getLastRow(), r = 0;
  if (last >= 2) { var f = sh.getRange(2, 17, last - 1, 1).createTextFinder(path).matchEntireCell(true).findNext(); if (f) r = f.getRow(); }
  if (r) sh.getRange(r, 1, 1, row.length).setValues([row]); else sh.appendRow(row);
}
function syncIndex_(uid, d) {
  var ip = 'intakes/' + uid, idx = rawGet_(ip) || {owner:uid, projects:[]};
  var list = (idx.projects || []).filter(function(x){ return x.id !== d.id; });

  list.sort(function(a, b){ return a.updatedAt < b.updatedAt ? 1 : -1; });
  idx.projects = list; idx.updatedAt = now_();
  rawSet_(ip, idx);
}

function pickReviewer_(d) {
  var list = rows_('Reviewers').filter(function(r){ return r.Email && String(r.Active).toUpperCase() !== 'FALSE'; });
  var types = scopesOf_(d);
  list = list.filter(function(r){ var t = String(r['Project types (blank = all)'] || '').toLowerCase(); return !t || types.some(function(k){ return t.indexOf(k) >= 0 || t.indexOf(SCOPES[k].toLowerCase()) >= 0; }); });
  if (!list.length) return {email:digestTo_()};

}

function payCheckout_(s, path, kind) {
  var p = parts_(path); if (!canRead_(s, p) || !isProjectPath_(p)) throw new Error('Not allowed.');
  var d = rawGet_(p.join('/')); if (!d) throw new Error('No such project');
  if (d.tier === 't1') return {url:'https://permitready.lemonsqueezy.com/checkout/buy/f091900b-7226-460b-8305-c25e2d422423'};
  var price = priceOf_(d), dep = Math.round(price * DEPOSIT_RATE), amt = kind === 'balance' ? price - dep : dep;
  if (kind === 'balance' && d.status !== 'ready') throw new Error('The balance is due once your drawings are ready.');
  if (kind !== 'balance' && ['screened'].indexOf(d.status) < 0) throw new Error('This project is not waiting for a deposit.');
  var c = d.contact || {};
  var body = {data:{type:'checkouts', attributes:{
      custom_price: amt * 100,
      product_options:{name:'PermitSetReady ' + (d.server && d.server.psr || '') + ' — ' + (kind === 'balance' ? 'balance' : 'deposit'), description:packageName_(d) + ' · ' + ((d.property || {}).address || ''), redirect_url: site_() + '/start/#project'},
      checkout_data:{email:c.email || s.email, name:c.name || '', custom:{job_path:p.join('/'), kind:kind === 'balance' ? 'balance' : 'deposit'}}},
    relationships:{store:{data:{type:'stores', id:String(prop_('LS_STORE_ID'))}}, variant:{data:{type:'variants', id:String(prop_('LS_VARIANT_ID'))}}}}};
  var res = UrlFetchApp.fetch('https://api.lemonsqueezy.com/v1/checkouts', {method:'post', contentType:'application/vnd.api+json', headers:{Accept:'application/vnd.api+json', Authorization:'Bearer ' + prop_('LS_API_KEY')}, payload:JSON.stringify(body), muteHttpExceptions:true});
  if (res.getResponseCode() >= 300) { log_('error', 'checkout', res.getContentText()); throw new Error('Payment page could not be created. Please try again shortly.'); }
  return {url:JSON.parse(res.getContentText()).data.attributes.url, amount:amt};
}

function lsWebhook_(e) {
  if (!prop_('LS_WEBHOOK_KEY') || e.parameter.key !== prop_('LS_WEBHOOK_KEY')) return {ok:false};
  var body = JSON.parse(e.postData.contents || '{}'), meta = body.meta || {}, data = body.data || {}, at = data.attributes || {};
  var custom = meta.custom_data || {};
  sh_('Orders').appendRow([new Date(), meta.event_name || '', data.id || '', at.user_email || '', at.first_order_item ? at.first_order_item.product_name : '', at.total_formatted || '', custom.job_path || '', custom.kind || '']);
  if (meta.event_name !== 'order_created' || !custom.job_path) return {ok:true};
  // Never trust the webhook body alone: read the order back from Lemon Squeezy.
  var res = UrlFetchApp.fetch('https://api.lemonsqueezy.com/v1/orders/' + encodeURIComponent(data.id), {headers:{Accept:'application/vnd.api+json', Authorization:'Bearer ' + prop_('LS_API_KEY')}, muteHttpExceptions:true});
  if (res.getResponseCode() !== 200) { log_('warn', 'webhook', 'order lookup failed ' + data.id); return {ok:true}; }
  var order = JSON.parse(res.getContentText()).data.attributes;
  if (order.status !== 'paid') return {ok:true};
  var lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    var path = custom.job_path, d = rawGet_(path); if (!d) return {ok:true};
    var rec = {orderId:data.id, paidAt:now_(), total:order.total, totalFormatted:order.total_formatted};
    if (custom.kind === 'balance') setStatus_(path, 'delivered', {payment:{balance:rec}});
    else if (d.status === 'screened') setStatus_(path, 'confirmed', {payment:{deposit:rec}});
    else { d.payment = d.payment || {}; d.payment.deposit = rec; rawSet_(path, d); }
  } finally { lock.releaseLock(); }
  return {ok:true};
}

// ---------------------------------------------------------------- drafting runner API
function runner_(req) {
  if (!prop_('RUNNER_KEY') || req.key !== prop_('RUNNER_KEY')) throw new Error('Not allowed.');
  if (req.action === 'runner.jobs') {
    // Jobs ready for the runner: deposit in and (no measuring, or as-built received), or City comments to address, or sent back by review.
    var out = [];
    rows_('Jobs').forEach(function(r){
      var d = rawGet_(r.Path); if (!d) return;
      var s = d.server || {}, ready = (d.status === 'confirmed' && (!(d.measure && d.measure.want) || s.asbuiltReceived)) || d.status === 'revising' || (d.status === 'drafting' && s.sentBack);
      if (ready) out.push({path:r.Path, psr:s.psr, status:d.status, folderId:s.folderId, folderUrl:s.folderUrl, package:packageName_(d), tier:d.tier, scopes:scopesOf_(d), address:(d.property || {}).address, municipality:(d.property || {}).municipality, reviewNotes:(d.admin || {}).reviewNotes || '', needsEngineer:!!(d.admin || {}).needsEngineer});
    });
    return {jobs:out};
  }
  if (req.action === 'runner.get') return {data:rawGet_(req.path)};
  if (req.action === 'runner.update') {
    var lock = LockService.getScriptLock(); lock.waitLock(20000);
    try {
      var extra = {server:{runnerNote:req.note || '', runnerAt:now_(), sentBack:false}};
      if (req.needsEngineer) extra.admin = {needsEngineer:true};
      var d2 = setStatus_(req.path, req.status, extra);
      return {status:d2.status};
    } finally { lock.releaseLock(); }
  }
  throw new Error('Unknown runner action');
}

// ---------------------------------------------------------------- watchers and digest
function hourlyWatch() {
  rows_('Jobs').forEach(function(r){
    try {
      var d = rawGet_(r.Path); if (!d || !d.server || !d.server.folderId) return;
      var folder = DriveApp.getFolderById(d.server.folderId);
      if (d.status === 'measuring') {
        var since = new Date(lastHistory_(d, 'measuring') || 0);
        if (newFiles_(subfolder_(folder, 'Survey'), since) || newFiles_(subfolder_(folder, 'Received'), since, /as.?built/i)) {
          setStatus_(r.Path, 'confirmed', {server:{asbuiltReceived:true}});
        }
      }
      if (d.status === 'consultant') {
        var since2 = new Date(lastHistory_(d, 'consultant') || 0);
        if (newFiles_(subfolder_(subfolder_(folder, 'Consultant'), 'Returned'), since2)) setStatus_(r.Path, 'ready');
      }
    } catch (e) { log_('warn', 'hourlyWatch ' + r.PSR, e); }
  });
}

function newFiles_(folder, since, rx) { var it = folder.getFiles(); while (it.hasNext()) { var f = it.next(); if (f.getDateCreated() > since && (!rx || rx.test(f.getName()))) return true; } return false; }

      subject = 'We received your project (' + (s.psr || 'PermitSetReady') + ')';
      body = '<p>Hi ' + esc_(first) + ',</p><p>Thanks for sending your project at ' + addr + '. Your package is <b>' + pkg + '</b> at a fixed price of <b>' + money_(price) + '</b> + HST.</p><p>To get started, pay the 40% deposit (' + money_(dep) + ') from your project page. You can follow every step there.</p>' + button_('Open my project', portal);
    } else if (type === 'deposit') {
      subject = 'Deposit received — your drawings are underway (' + (s.psr || '') + ')';
      body = '<p>Hi ' + esc_(first) + ',</p><p>Your deposit is in. ' + (d.measure && d.measure.want ? 'Our measuring partner will contact you to book the visit; drafting starts once the house is measured.' : 'Drafting starts now. Typical delivery is 7–10 business days.') + '</p>' + button_('Follow my project', portal);
    } else if (type === 'measuring_client') {
      subject = 'Booking your measuring visit (' + (s.psr || '') + ')';
      body = '<p>Hi ' + esc_(first) + ',</p><p>Our measuring partner will be in touch in the next few business days to book a time to measure the house at ' + addr + '. You picked: ' + esc_((d.brief || {}).measureWhen || 'any time') + '.</p>' + button_('Follow my project', portal);
    } else if (type === 'measuring_partner') {
      var mp = pickConsultant_(null, 'measuring'); to = mp ? mp.Email : digestTo_();
      subject = 'Measuring request: ' + ((d.property || {}).address || '') + ', ' + ((d.property || {}).municipality || '') + ' (' + (s.psr || '') + ')';
      body = '<p>Hello' + (mp ? ' ' + esc_(String(mp.Name).split(' ')[0]) : '') + ',</p><p>Please measure the existing house for a new project:</p><ul><li>Address: ' + addr + ', ' + esc_((d.property || {}).municipality || '') + '</li><li>Size: about ' + esc_((d.measure || {}).sqft || '?') + ' sq ft</li><li>Scope: ' + pkg + '</li><li>Client: ' + esc_(c.name || '') + ', ' + esc_(c.phone || '') + ', ' + esc_(c.email || '') + '</li><li>Preferred time: ' + esc_((d.brief || {}).measureWhen || 'any') + '</li></ul><p>Please upload the as-built to this folder when done: <a href="' + s.folderUrl + '">' + esc_(s.psr || 'job folder') + '</a> (Survey subfolder), or reply with a link.</p>';
    } else if (type === 'consultant_transmittal') {
      to = consultant.Email;
      subject = (s.psr || 'PermitSetReady') + ': drawings for review and stamping — ' + ((d.property || {}).address || '');
      body = '<p>Hello ' + esc_(String(consultant.Name).split(' ')[0]) + ',</p><p>Please review and stamp the attached set for ' + addr + ', ' + esc_((d.property || {}).municipality || '') + ' (' + pkg + ').</p><p>Drawings: <a href="' + s.folderUrl + '">job folder</a> (PDF subfolder). Please return the stamped set to the Consultant › Returned subfolder, or reply with your comments and fee.</p>';
    } else if (type === 'consultant_client') {
      subject = 'Your drawings are with the engineer (' + (s.psr || '') + ')';
      body = '<p>Hi ' + esc_(first) + ',</p><p>Your drawings passed our review and are now with the engineer for review and stamping. We\'ll let you know as soon as they\'re back.</p>' + button_('Follow my project', portal);
    } else if (type === 'ready') {
      subject = 'Your permit drawings are ready (' + (s.psr || '') + ')';
      body = '<p>Hi ' + esc_(first) + ',</p><p>Your drawings for ' + addr + ' are ready. Pay the balance (' + money_(price - dep) + ' + HST) on your project page and the download unlocks right away, with your submission checklist.</p>' + button_('Pay and download', portal);
    } else if (type === 'delivered') {
      subject = 'Download your permit set (' + (s.psr || '') + ')';
      body = '<p>Hi ' + esc_(first) + ',</p><p>Thank you. Your permit set is ready to download from your project page, along with the checklist for applying to ' + esc_((d.property || {}).municipality || 'the City') + '.</p><p>If the examiner sends comments, upload them on the same page and we\'ll revise the drawings (one round included).</p>' + button_('Download my drawings', portal);
    } else if (type === 'revised') {
      subject = 'Your revised drawings are ready (' + (s.psr || '') + ')';
      body = '<p>Hi ' + esc_(first) + ',</p><p>We\'ve revised your drawings for the examiner\'s comments. Download them from your project page and resubmit.</p>' + button_('Download revised drawings', portal);
    } else return;
    if (!to) return;
    var html = shell_(subject.replace(/ \(PSR-\d+\)$/, ''), body);
    var mode = (setting_('SEND_MODE_' + type) || 'draft').toLowerCase();
    if (mode === 'send') GmailApp.sendEmail(to, subject, html.replace(/<[^>]+>/g, ' '), fromOpts_({htmlBody:html}));
    else {
      var draft = GmailApp.createDraft(to, subject, html.replace(/<[^>]+>/g, ' '), fromOpts_({htmlBody:html}));
      var lab = GmailApp.getUserLabelByName('PSR-Auto') || GmailApp.createLabel('PSR-Auto');
      try { draft.getMessage().getThread().addLabel(lab); } catch (e) {}
    }
    sh_('Emails').appendRow([new Date(), type, to, subject, mode, s.psr || path]);
  } catch (err) { log_('error', 'email ' + type, err); }
}
function sendInternal_(to, subject, inner) {
  try { GmailApp.sendEmail(to, subject, inner.replace(/<[^>]+>/g, ' '), fromOpts_({htmlBody:shell_(subject, inner)})); } catch (e) { log_('error', 'internal email', e); }
}
